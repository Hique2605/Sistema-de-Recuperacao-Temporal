# 1. Libs de sistema
sudo apt update
sudo apt install -y python3 python3-venv python3-pip ffmpeg libgl1 libglib2.0-0

# 2. Extrair o projeto
unzip sistema_gravacao.zip -d sistema_gravacao
cd sistema_gravacao

# 3. Ambiente virtual
python3 -m venv venv
source venv/bin/activate

# 4. Dependências Python (base)
pip install --upgrade pip
pip install -r requirements.txt

# 5. Dependências do botão GPIO
pip install -r requirements-rpi.txt

# 6. Configurar câmera, pino do botão e (opcional) e-mail/retenção
cp .env.example .env
nano .env   # preencher CAMERA_IP, CAMERA_USUARIO, CAMERA_SENHA, GPIO_SALVAR_PIN
            # opcional: EMAIL_ATIVADO + dados SMTP, RETENCAO_DIAS

# 7. Rodar
python -m src.main
